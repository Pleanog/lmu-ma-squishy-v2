@echo off
echo ===================================================
echo Starte Server-Umgebung
echo ===================================================
echo.

echo [1/3] Starte PocketBase im Hintergrund...
:: Öffnet ein neues Fenster, wechselt in den Ordner und startet PB
start "PocketBase Database" cmd /k "cd backend\pocketbase_0.35.1_windows_amd64 && pocketbase serve"

echo Warte 2 Sekunden, damit die Datenbank sicher hochgefahren ist...
timeout /t 2 /nobreak >nul

echo [2/3] Starte Vue Frontend in neuem Fenster...
:: Öffnet ein neues Fenster für Vite
start "Vue Frontend" cmd /k "cd frontend && npm run dev"

echo [3/3] Starte FastAPI Backend hier im Hauptfenster...
:: Wechselt im aktuellen Fenster in den Server-Ordner, aktiviert venv und startet uvicorn
cd backend\server
call venv\Scripts\activate.bat
uvicorn main:app --reload --host 0.0.0.0 --port 8000s
